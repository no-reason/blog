### 正交多项式根的性质

$
\begin{aligned}

&\\

\end{aligned}
$

### 正交多项式系的三项递推

$
\begin{aligned}
\begin{cases}
\hat{\phi_{0}}(x)=1\\
\hat{\phi_{1}}(x)=x-\frac{(x,1)}{(1,1)}\\
\hat{\phi_{n+1}}(x)=(x-a_{n})\hat{\phi_{n}}(x)-b_{n-1}\hat{\phi_{n-1}}(x)\\
&\\
a_{n}=\frac{(x,\hat{\phi_{n}})}{(\hat{\phi_{n}},\hat{\phi_{n}})}\\
b_{n}=\frac{(\hat{\phi_{n}},\hat{\phi_{n}})}{(\hat{\phi_{n-1}},\hat{\phi_{n-1}})}\\
\end{cases}\\
&\\\\
&
\end{aligned}
$

#### 常用的正交多项式系

##### Legendre多项式系
$
\begin{aligned}
&[-1,1]上以\rho(x)=1位权函数的正交多项式系\\
&P_{0}(x)=1,P_{n}(x)=\frac{1}{2^{n}n!}\frac{d^{n}}{dx^{n}}(x^{2}-1)^{n}\\
&P_{n}(x)的首项系数为\frac{(2n)!}{2^{n}(n!)^{2}}\\
&正交性:\\
&(P_{n},P_{m})=\int_{-1}^{1}P_{n}(x)P_{m}(x)dx=\begin{cases}
0,n\ne m\\  
\frac{2}{2n+1},n=m
\end{cases}\\
&奇偶性:\\
&P_{n}(x)=(-1)^{n}P_{n}(-x)\\
&三项递推公式:\\
&\begin{cases}
P_{0}(x)=1\\
P_{1}(x)=x\\
P_{n+1}(x)=\frac{2n+1}{n+1}xP_{n}(x)-\frac{n}{n+1}P_{n-1}(x)\\
\end{cases}
\end{aligned}
$
##### (第一类)Chebyshev多项式系

$
\begin{aligned}
&是定义在[-1,1]上以\rho(x)=\frac{1}{\sqrt{1-x^{2}}}为权函数的正交多项式系\\
&T_{n}(x)=cos(narccosx)\\
&T_{n}(x)的首项系数为2^{n-1}\\
&正交性:\\
&(T_{n},T_{m})=\int_{-1}^{1}\frac{  T_{n}(x)T_{m}(x)}{\sqrt{1-x^{2}}}dx=\begin{cases}
0,n\ne m\\  
\pi,n=m=0\\
\frac{\pi}{2},n=m\ne 0
\end{cases}\\
&奇偶性:\\
&T_{n}(x)=(-1)^{n}T_{n}(-x)\\
&三项递推公式:\\
&\begin{cases}
T_{0}(x)=1\\    
T_{1}(x)=x\\
T_{n+1}(x)=2xT_{n}(x)-T_{n-1}(x)
\end{cases}\\
&
\end{aligned}
$

##### (第二类)Chebyshev多项式系
$
\begin{aligned}
&是定义在[-1,1]上以\rho(x)=\sqrt{1-x^{2}}为权函数的正交多项式系\\
&U_{n}(x)=\frac{sin((n+1)arccosx)}{sin(arccosx)}\\
&U_{n}(x)的首项系数为2^{n}\\
&正交性:\\
&(U_{n},U_{m})=\int_{-1}^{1}U_{n}(x)U_{m}(x)\sqrt{1-x^{2}}dx=\begin{cases}
0,n\ne m\\  
\frac{\pi}{2},n=m
\end{cases}\\
&奇偶性:\\
&U_{n}(x)=(-1)^{n}U_{n}(-x)\\
&三项递推公式:\\
&\begin{cases}
U_{0}(x)=1\\    
U_{1}(x)=2x\\
U_{n+1}(x)=2xU_{n}(x)-U_{n-1}(x)
\end{cases}\\
&
\end{aligned}
$
##### Laguerre多项式系
$
\begin{aligned}
&是定义在[0,+\infty)上以\rho(x)=e^{-x}为权函数的正交多项式系\\
&L_{n}(x)=e^{x}\frac{d^{n}}{dx^{n}}(x^{n}e^{-x})\\
&L_{n}(x)的首项系数为(-1)^{n}\\
&正交性:\\
&(L_{n},L_{m})=\int_{0}^{+\infty}L_{n}(x)L_{m}(x)e^{-x}dx=\begin{cases}
(n!)^{2},n\ne m\\  
1,n=m                           
\end{cases}\\
&三项递推公式:\\
&\begin{cases}
L_{0}(x)=1\\    
L_{1}(x)=1-x\\
L_{n+1}(x)=(2n+1-x)L_{n}(x)-n^{2}L_{n-1}(x)
\end{cases}\\
&
\end{aligned}
$

##### Hermite多项式系

$$
\begin{aligned}
&是定义在(-\infty,+\infty)上以\rho(x)=e^{-x^{2}}为权函数的正交多项式系\\
&H_{n}(x)=(-1)^{n}e^{x^{2}}\frac{d^{n}}{dx^{n}}(e^{-x^{2}})\\
&H_{n}(x)的首项系数为2^{n}\\
&正交性:\\
&(H_{n},H_{m})=\int_{-\infty}^{+\infty}H_{n}(x)H_{m}(x)e^{-x^{2}}dx=\begin{cases}
0,n\ne m\\  
2^{n}n!\sqrt{\pi},n=m
\end{cases}\\
&奇偶性:\\
&H_{n}(x)=(-1)^{n}H_{n}(-x)\\
&三项递推公式:\\
&\begin{cases}
H_{0}(x)=1\\    
H_{1}(x)=2x\\
H_{n+1}(x)=2xH_{n}(x)-2nH_{n-1}(x)
\end{cases}\\
&
\end{aligned}
$$
